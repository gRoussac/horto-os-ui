5fn systemctl(ctx: &mut HostContext, args: &[&str]) -> Result<()> {
6    let summary = format!("systemctl {}", args.join(" "));
7    if ctx.is_dry_run() {
8        ctx.plan_action(summary);
9        return Ok(());
10    }
11    ctx.log(&summary);
12    let status = Command::new("systemctl")
13        .args(args)
14        .status()
15        .map_err(|e| HortoError::command("systemctl", e.to_string()))?;
16    if !status.success() {
17        return Err(HortoError::command(
18            "systemctl",
19            format!("{} -> {status}", args.join(" ")),
20        ));
21    }
22    Ok(())
23}

86pub fn unit_present(unit: &str) -> bool {
87    Command::new("systemctl")
88        .args(["list-unit-files", &format!("{unit}.service")])
89        .output()
90        .ok()
91        .map(|o| o.status.success() && !o.stdout.is_empty())
92        .unwrap_or(false)
