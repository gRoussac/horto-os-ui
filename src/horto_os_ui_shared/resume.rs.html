9#[derive(Debug, Clone, Serialize, Deserialize, Default)]
10pub struct ResumeState {
11    pub steps: Vec<ResumeStepRecord>,
12}
13
14#[derive(Debug, Clone, Serialize, Deserialize)]
15pub struct ResumeStepRecord {
16    pub id: String,
17    pub step_version: u32,
18    pub completed_at: DateTime<Utc>,
19    pub ok: bool,
20}
21
22#[derive(Debug, Clone, Copy, PartialEq, Eq)]
23pub enum StepStatus {
24    Pending,
25    Done,
26    Stale,
27    Failed,
28}

30pub fn load(path: &Path) -> Result<ResumeState> {
31    if !path.exists() {
32        return Ok(ResumeState::default());
33    }
34    let text = fs::read_to_string(path)?;
35    Ok(serde_json::from_str(&text)?)
36}
37
38pub fn save(path: &Path, state: &ResumeState) -> Result<()> {
39    if let Some(parent) = path.parent() {
40        fs::create_dir_all(parent)?;
41    }
42    let text = serde_json::to_string_pretty(state)?;
43    fs::write(path, text)?;
44    Ok(())
45}

47pub fn record_ok(paths: &HostPaths, step: &dyn Step) -> Result<()> {
48    let path = paths.resume_file();
49    let mut state = load(&path)?;
50    state.steps.retain(|r| r.id != step.id());
51    state.steps.push(ResumeStepRecord {
52        id: step.id().to_string(),
53        step_version: step.step_version(),
54        completed_at: Utc::now(),
55        ok: true,
56    });
57    save(&path, &state)
58}
59
60pub fn status_for(state: &ResumeState, step: &dyn Step) -> StepStatus {
61    match state.steps.iter().rev().find(|r| r.id == step.id()) {
62        Some(r) if r.ok && r.step_version >= step.step_version() => StepStatus::Done,
63        Some(r) if r.ok && r.step_version < step.step_version() => StepStatus::Stale,
64        Some(r) if !r.ok => StepStatus::Failed,
65        _ => StepStatus::Pending,
66    }
67}

69/// True when recorded completion is older than the binary's step_version.
70pub fn is_stale(recorded_version: u32, binary_version: u32) -> bool {
71    binary_version > recorded_version
72}
73
74pub fn mark_failed(paths: &HostPaths, step_id: &str, step_version: u32) -> Result<()> {
75    let path = paths.resume_file();
76    let mut state = load(&path).unwrap_or_default();
77    state.steps.retain(|r| r.id != step_id);
78    state.steps.push(ResumeStepRecord {
79        id: step_id.to_string(),
80        step_version,
81        completed_at: Utc::now(),
82        ok: false,
83    });
84    save(&path, &state)
85}
