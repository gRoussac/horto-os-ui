12const OS_REQUIRED: &[&str] = &[
13    "MY_HOSTNAME",
14    "OS_TYPE",
15    "NPU_TYPE",
16    "INSTALL_TYP",
17    "IOT_LAN",
18];
19const IOT_REQUIRED: &[&str] = &["WIFI_INTERFACE", "WIFI_SSID"];
20const IOT_PACKAGES: &[&str] = &["hostapd", "dnsmasq", "iptables", "avahi-daemon"];

22impl Step for S2Env {
23    fn id(&self) -> &'static str {
24        "s2"
25    }
26    fn title(&self) -> &'static str {
27        "Configure environment variables"
28    }
29    fn reference_script(&self) -> &'static str {
30        "s2_init_env_vars.sh"
31    }
32    fn step_version(&self) -> u32 {
33        2
34    }
35    fn depends_on(&self) -> &'static [&'static str] {
36        &["s1"]
37    }
38    fn is_done(&self, ctx: &HostContext) -> bool {
39        let os_path = ctx.paths.os_configuration_file();
40        let Ok(os_map) = envfile::load(&os_path) else {
41            return false;
42        };
43        if envfile::require_keys(&os_map, OS_REQUIRED).is_err() {
44            return false;
45        }
46        if wants_iot_lan(&os_map) {
47            let iot_path = ctx.paths.iot_lan_env_file();
48            return envfile::load(&iot_path)
49                .ok()
50                .and_then(|m| envfile::require_keys(&m, IOT_REQUIRED).ok())
51                .is_some();
52        }
53        true
54    }
55    fn plan(&self, ctx: &mut HostContext) -> Result<Vec<PlannedAction>> {
56        ctx.plan_action(format!(
57            "create/update {} from embedded config/os-configuration.env",
58            ctx.paths.os_configuration_file().display()
59        ));
60        ctx.plan_action("prompt OS_TYPE, NPU_TYPE, INSTALL_TYP, IOT_LAN, hostname, URL");
61        ctx.plan_action("if IOT_LAN=y: apt IoT packages + iot-lan_conf.env + ETH discovery");
62        Ok(ctx.planned.clone())
63    }
64    fn apply(&self, ctx: &mut HostContext) -> Result<()> {
65        fs::ensure_dir(ctx, &ctx.paths.active_setup.clone())?;
66        let os_active = ctx.paths.os_configuration_file();
67        let mut os_map = load_or_embed(ctx, &os_active, "config/os-configuration.env")?;
68
69        if ctx.is_dry_run() {
70            ctx.plan_action(format!("write {}", os_active.display()));
71            ctx.plan_action("optional IoT-LAN env + packages when IOT_LAN=y");
72            return Ok(());
73        }
74
75        prompt_os_conf(ctx, &mut os_map)?;
76        envfile::require_keys(&os_map, OS_REQUIRED)?;
77        envfile::write(&os_active, &os_map)?;
78        ctx.log(format!("Saved OS configuration to {}", os_active.display()));
79
80        if wants_iot_lan(&os_map) {
81            apply_iot_lan(ctx)?;
82        } else {
83            ctx.log("setup without IOT_LAN detected");
84        }
85        Ok(())
86    }
87}

89fn load_or_embed(
90    ctx: &HostContext,
91    active: &std::path::Path,
92    embed_path: &str,
93) -> Result<BTreeMap<String, String>> {
94    if active.exists() && !ctx.is_dry_run() {
95        return envfile::load(active);
96    }
97    let template =
98        embed::get_str(embed_path).ok_or_else(|| HortoError::EmbedMissing(embed_path.into()))?;
99    Ok(envfile::parse(&template))
100}

102fn prompt_os_conf(ctx: &mut HostContext, map: &mut BTreeMap<String, String>) -> Result<()> {
103    let hostname = ctx.prompt(
104        "Device hostname",
105        map.get("MY_HOSTNAME")
106            .map(String::as_str)
107            .unwrap_or("Horto-OS_xxx"),
108    );
109    let os_type = ctx.prompt(
110        "OS type (debian/armbian)",
111        map.get("OS_TYPE").map(String::as_str).unwrap_or("debian"),
112    );
113    let npu = ctx.prompt(
114        "NPU type (rkRK3576/rkRK3588/...)",
115        map.get("NPU_TYPE")
116            .map(String::as_str)
117            .unwrap_or("rkRK3588"),
118    );
119    let ram = ctx.prompt(
120        "RAM size label",
121        map.get("RAM_SYZE").map(String::as_str).unwrap_or("8gb"),
122    );
123    let install = ctx.prompt(
124        "Install type (home/satellite/hortex)",
125        map.get("INSTALL_TYP").map(String::as_str).unwrap_or("home"),
126    );
127    let iot = ctx.prompt(
128        "Enable IOT-LAN (y/n)",
129        map.get("IOT_LAN").map(String::as_str).unwrap_or("n"),
130    );
131    let my_url = ctx.prompt(
132        "Public URL / domain",
133        map.get("MY_URL")
134            .map(String::as_str)
135            .unwrap_or("YourDomainName.net"),
136    );
137    let cf = ctx.prompt(
138        "Cloudflare token (optional)",
139        map.get("MY_CLOUDFLARE_TOKEN")
140            .map(String::as_str)
141            .unwrap_or(""),
142    );
143
144    envfile::set_key(map, "MY_HOSTNAME", hostname);
145    envfile::set_key(map, "OS_TYPE", os_type);
146    envfile::set_key(map, "NPU_TYPE", npu);
147    envfile::set_key(map, "RAM_SYZE", ram);
148    envfile::set_key(map, "INSTALL_TYP", install);
149    envfile::set_key(map, "IOT_LAN", normalize_yn(&iot));
150    envfile::set_key(map, "MY_URL", my_url);
151    envfile::set_key(map, "MY_CLOUDFLARE_TOKEN", cf);
152    Ok(())
153}

155fn apply_iot_lan(ctx: &mut HostContext) -> Result<()> {
156    ctx.log("IOT-LAN setup is next");
157    ctx.log("Installing IoT LAN components...");
158    if crate::context::is_root() {
159        apt::apt_install(ctx, IOT_PACKAGES)?;
160        ctx.log("Base packages for IOT-LAN installed.");
161    } else {
162        ctx.log("Not root; skipping IoT apt install");
163    }
164
165    let iot_active = ctx.paths.iot_lan_env_file();
166    let mut map = load_or_embed(ctx, &iot_active, "config/iot-lan_conf.env")?;
167
168    let wifi_if = ctx.prompt(
169        "WiFi interface",
170        map.get("WIFI_INTERFACE")
171            .map(String::as_str)
172            .unwrap_or("wlx0_xxxxx"),
173    );
174    let wifi_ssid = ctx.prompt(
175        "WiFi SSID",
176        map.get("WIFI_SSID")
177            .map(String::as_str)
178            .unwrap_or("Horto-IoT-LAN"),
179    );
180    let wifi_pass = ctx.prompt(
181        "WiFi passphrase",
182        map.get("WIFI_PASSPHRASE").map(String::as_str).unwrap_or(""),
183    );
184
185    envfile::set_key(&mut map, "WIFI_INTERFACE", wifi_if);
186    envfile::set_key(&mut map, "WIFI_SSID", wifi_ssid);
187    envfile::set_key(&mut map, "WIFI_PASSPHRASE", wifi_pass);
188    discover_eth(&mut map, ctx);
189    envfile::require_keys(&map, IOT_REQUIRED)?;
190    envfile::write(&iot_active, &map)?;
191    ctx.log(format!(
192        "Saved IoT-LAN variables to {}",
193        iot_active.display()
194    ));
195    Ok(())
196}

198fn wants_iot_lan(map: &BTreeMap<String, String>) -> bool {
199    map.get("IOT_LAN")
200        .map(|v| matches!(v.trim().to_ascii_lowercase().as_str(), "y" | "yes"))
201        .unwrap_or(false)
202}

204fn normalize_yn(raw: &str) -> String {
205    match raw.trim().to_ascii_lowercase().as_str() {
206        "y" | "yes" => "y".into(),
207        _ => "n".into(),
208    }
209}

211fn discover_eth(map: &mut BTreeMap<String, String>, ctx: &mut HostContext) {
212    let ifaces = list_en_ifaces();
213    let (eth0, eth1, eth2) = pick_eth(&ifaces);
214    envfile::set_key(map, "ETH_LAN", eth0.clone());
215    envfile::set_key(map, "ETH_IOT1", eth1.clone());
216    if let Some(e2) = eth2 {
217        envfile::set_key(map, "ETH_IOT2", e2.clone());
218        ctx.log(format!(
219            "Discovered ETH_LAN={eth0}, ETH_IOT1={eth1}, ETH_IOT2={e2}"
220        ));
221    } else {
222        ctx.log(format!(
223            "Discovered ETH_LAN={eth0}, ETH_IOT1={eth1}, ETH_IOT2=not-set"
224        ));
225    }
226}

228fn list_en_ifaces() -> Vec<(String, bool)> {
229    let output = Command::new("ip").args(["-o", "link", "show"]).output();
230    let Ok(output) = output else {
231        return Vec::new();
232    };
233    let text = String::from_utf8_lossy(&output.stdout);
234    let mut out = Vec::new();
235    for line in text.lines() {
236        let Some(rest) = line.split_once(": ").map(|(_, r)| r) else {
237            continue;
238        };
239        let name = rest.split(':').next().unwrap_or("").trim();
240        if !name.starts_with("en") {
241            continue;
242        }
243        let lower_up = line.contains("LOWER_UP");
244        out.push((name.to_string(), lower_up));
245    }
246    out
247}

249fn pick_eth(ifaces: &[(String, bool)]) -> (String, String, Option<String>) {
250    let active: Vec<&str> = ifaces
251        .iter()
252        .filter(|(_, up)| *up)
253        .map(|(n, _)| n.as_str())
254        .collect();
255    if !active.is_empty() {
256        let eth0 = active[0].to_string();
257        let rest: Vec<&str> = ifaces
258            .iter()
259            .map(|(n, _)| n.as_str())
260            .filter(|n| *n != eth0)
261            .collect();
262        let eth1 = rest.first().unwrap_or(&"lan1").to_string();
263        let eth2 = rest.get(1).map(|s| (*s).to_string());
264        return (eth0, eth1, eth2);
265    }
266    let names: Vec<&str> = ifaces.iter().map(|(n, _)| n.as_str()).collect();
267    let eth0 = names.first().unwrap_or(&"wan").to_string();
268    let eth1 = names.get(1).unwrap_or(&"lan1").to_string();
269    let eth2 = names.get(2).map(|s| (*s).to_string());
270    (eth0, eth1, eth2)
271}
