11    pub ip: String,
12    pub mac: String,
13    pub expires: String,

16pub fn read_leases(lease_file: &Path, leases_json: &Path) -> Vec<LeaseEntry> {
17    if lease_file.is_file() {
18        if let Ok(text) = std::fs::read_to_string(lease_file) {
19            return parse_dnsmasq_leases(&text);
20        }
21    }
22    if leases_json.is_file() {
23        if let Ok(text) = std::fs::read_to_string(leases_json) {
24            if let Ok(list) = serde_json::from_str::<Vec<LeaseEntry>>(&text) {
25                return list;
26            }
27        }
28    }
29    Vec::new()
30}

32pub fn parse_dnsmasq_leases(text: &str) -> Vec<LeaseEntry> {
33    let mut out = Vec::new();
34    for line in text.lines() {
35        let parts: Vec<&str> = line.split_whitespace().collect();
36        if parts.len() < 4 {
37            continue;
38        }
39        out.push(LeaseEntry {
40            expires: parts[0].to_string(),
41            mac: parts[1].to_string(),
42            ip: parts[2].to_string(),
43            hostname: parts[3].to_string(),
44        });
45    }
46    out
47}

49/// Port of export_dhcp_leases.sh: write leases.json + leases.html under docker assets.
50pub fn export_dhcp_leases(ctx: &mut HostContext) -> Result<()> {
51    let lease_file = ctx.paths.lease_file.clone();
52    let out_dir = ctx.paths.docker_assets();
53    let out_json = ctx.paths.leases_json();
54    let out_html = ctx.paths.leases_html();
55
56    fs_kit::ensure_dir(ctx, &out_dir)?;
57
58    if lease_file.is_file() && out_html.is_file() {
59        if let (Ok(lm), Ok(hm)) = (
60            lease_file.metadata().and_then(|m| m.modified()),
61            out_html.metadata().and_then(|m| m.modified()),
62        ) {
63            if lm <= hm {
64                ctx.log("Leases unchanged since last export; skipping.");
65                return Ok(());
66            }
67        }
68    }
69
70    if !lease_file.is_file() {
71        let empty_json = "[]\n";
72        let empty_html = "<html><body><p>No DHCP leases found.</p></body></html>\n";
73        fs_kit::write_file(ctx, &out_json, empty_json.as_bytes())?;
74        fs_kit::write_file(ctx, &out_html, empty_html.as_bytes())?;
75        ctx.log("Lease file not found, wrote empty output.");
76        return Ok(());
77    }
78
79    if ctx.is_dry_run() {
80        ctx.plan_action(format!(
81            "export leases from {} to {} and {}",
82            lease_file.display(),
83            out_json.display(),
84            out_html.display()
85        ));
86        return Ok(());
87    }
88
89    let text = std::fs::read_to_string(&lease_file)
90        .map_err(|e| HortoError::msg(format!("read leases: {e}")))?;
91    let leases = parse_dnsmasq_leases(&text);
92    let json = serde_json::to_string_pretty(&leases)?;
93    let html = render_leases_html(&leases);
94    std::fs::write(&out_json, json)?;
95    std::fs::write(&out_html, html)?;
96    let _ = SystemTime::now();
97    ctx.log(format!(
98        "DHCP leases exported to {} and {}",
99        out_json.display(),
100        out_html.display()
101    ));
102    Ok(())
103}

105fn render_leases_html(leases: &[LeaseEntry]) -> String {
106    let mut out = String::from(
107        r#"<!DOCTYPE html>
108<html lang="en">
109<head>
110<meta charset="utf-8">
111<meta name="viewport" content="width=device-width, initial-scale=1">
112<style>
113body{font-family:ui-sans-serif,sans-serif;margin:0;padding:0;background:transparent;color:inherit}
114table{width:100%;border-collapse:collapse;font-size:.85rem}
115th,td{text-align:left;padding:6px 8px;border-bottom:1px solid rgba(255,255,255,.12)}
116th{font-weight:600;opacity:.7;text-transform:uppercase;font-size:.7rem;letter-spacing:.5px}
117tr:hover{background:rgba(255,255,255,.05)}
118</style>
119</head>
120<body>
121<table>
122<tr><th>Hostname</th><th>IP</th><th>MAC</th></tr>
123"#,
124    );
125    for l in leases {
126        out.push_str(&format!(
127            "<tr><td>{}</td><td>{}</td><td style=\"font-family:monospace;font-size:.8rem\">{}</td></tr>\n",
128            html_escape(&l.hostname),
129            html_escape(&l.ip),
130            html_escape(&l.mac)
131        ));
132    }
133    out.push_str("</table>\n</body>\n</html>\n");
134    out
135}

137fn html_escape(s: &str) -> String {
138    s.replace('&', "&amp;")
139        .replace('<', "&lt;")
140        .replace('>', "&gt;")
141        .replace('"', "&quot;")
142}
